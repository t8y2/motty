# End to end: the status icon's menu lists the saved hosts and workspaces of the
# window, and choosing one opens it against the loopback SSH fixture on port
# 2222. The menu is the system's, so it is read with statusmenu and used with
# statusselect, not clicked.
#
#   (cd tools/e2esshd && go build -o /tmp/e2esshd . && /tmp/e2esshd 2222) &
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/tray_saved_hosts_e2e.txt

# Save a host the fixture answers on. Its port starts at 22, so the field's
# text is selected and replaced.
click id:shell/add-host
type office
click id:shell/dialog-address
type 127.0.0.1
click id:shell/dialog-port
select 0 2
type 2222
click id:shell/dialog-user
type tester
click id:shell/dialog-submit
expect text "office" within 2000

# The menu lists the saved host under Hosts, without any session first. Choosing
# it opens a tab and dials: trust the fixture once, then answer its password.
statusmenu 1
statusselect 1 3.1
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
key enter
waitfor "tester@e2e" 8000
statusmenu 1

# Save a workspace template holding that host. The terminal page hides the
# sidebar, so go back to the hosts page first.
click id:shell/home
click id:shell/workspaces
click id:workspaces/new
click "Save template"
type Office
click id:workspaces/draft-search
type office
click id:workspaces/draft-host/#0
click 1049 717
expect text "Office" within 3000

# The menu lists the saved workspace too; choosing it opens another workspace
# with its own session, which shows in the count.
statusmenu 1
statusselect 1 5.1
statusmenu 1
