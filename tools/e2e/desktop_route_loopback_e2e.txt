# End to end: a VNC host reached through a jump route, with real SSH and a real
# desktop. The route signs in to the loopback SSH fixture and dials the VNC
# fixture's port from there; the VNC host is given no ViaSSH every-machine
# tunnel, so a picture can only appear if the route carried the connection.
#
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   go build -o /tmp/vncfixture ./tools/e2e/vncfixture && /tmp/vncfixture 5901 &
#   (cd tools/e2esshd && go build -o /tmp/e2esshd .) && /tmp/e2esshd 2222 127.0.0.1:5901 &
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/desktop_route_loopback_e2e.txt
#
# The SSH fixture's password is "secret"; the VNC fixture asks for none.
