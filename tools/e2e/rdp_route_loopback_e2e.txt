# End to end: an RDP host reached through a jump route, with a real SSH server
# (tools/e2esshd) and a real RDP server (tools/live/rdp). The route signs in to
# the SSH fixture and dials the RDP port from there; the RDP host has no SSH
# server of its own, so the picture can only appear if the route carried it.
#
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   docker build --platform linux/amd64 -t motty-e2e-rdp:latest tools/live/rdp
#   docker run -d --name motty-e2e-rdp -p 127.0.0.1:3389:3389 motty-e2e-rdp:latest
#   (cd tools/e2esshd && go build -o /tmp/e2esshd .) && /tmp/e2esshd 2222 127.0.0.1:3389 &
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/rdp_route_loopback_e2e.txt
#
# The SSH fixture's password is "secret"; the RDP account is motty / motty-pw.
