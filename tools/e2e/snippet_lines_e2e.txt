# lang: en
# End to end: a snippet can send its lines one at a time. Against the loopback
# SSH fixture:
#
#   (cd tools/e2esshd && go build -o /tmp/e2esshd . && E2E_ALLOW_EXEC=1 /tmp/e2esshd 22601 &)
#   go build -tags vivahead,dev -o /tmp/motty-head ./cmd/motty
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/snippet_lines_e2e.txt
#
# What to look for:
# - The snippet dialog has "Send multi-line commands one line at a time" under
#   "Press Enter after sending it".
# - With it ticked, the terminal shows "first-42" and then "second-48" as output
#   (the typed commands show $((6*7)); only the output shows the numbers).

click "Search"
type tester@127.0.0.1:22601
click "Connect"
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
key enter
waitfor "tester@e2e" 8000

click "Hosts"
click "Snippets"
click "Add snippet"
waitfor "Add a snippet" 3000
type Two lines
key tab
type echo first-$((6*7))
key enter
type echo second-$((6*8))
click "Send multi-line commands one line at a time"
shot /tmp/snippet-lines-dialog.png
click "Save"
click "Send to terminal"
expect text "first-42" within 4000
expect text "second-48" within 4000
shot /tmp/snippet-lines-sent.png
