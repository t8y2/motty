# double click, filters by its own search box, and closes with its shortcut.
#
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   (cd tools/e2esshd && go build -o /tmp/motty-e2esshd .) && /tmp/motty-e2esshd 2244 &
#   MOTTY_LANG=en python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/hosts_panel_e2e.txt
#
# The shortcut is Command+Shift+B on macOS (Control+Shift+B elsewhere); the
# headless build of the platform the script runs on takes the matching chord.
