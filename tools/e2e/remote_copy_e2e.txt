# Two loopback SSH servers, a file copied from A's file panel to B's file panel
# through Motty's memory. Start two e2esshd instances first:

#   /tmp/e2esshd 2233 &
# Write a file named server-copy-demo.txt into the first server's sftp root
# (printed at startup), then:
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   MOTTY_LANG=en python3 tools/headless_drive.py "$(mktemp -d)" \
#     tools/e2e/remote_copy_e2e.txt
click "Search"
