# End to end: change the quality of a running VNC session from the bar's Quality
# menu, against the loopback VNC fixture (go build -o /tmp/vncfixture
# ./tools/e2e/vncfixture && /tmp/vncfixture 5901 &). Its log then shows
# "client changed to 16 bits per pixel" with no new connection. Chinese labels;
# coordinates are those of the 1400x900 headless window.
#
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/vnc_quality_live_e2e.txt

type fixture-vnc

settle 500

settle 2500
