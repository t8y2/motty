# End to end: a jump route whose target is an RDP host. The route reaches the
# desktop port from its jump host, so the Windows machine needs no SSH server.
# No desktop server is started: this drives the host form and the route page up
# to the saved route and the form that shows it.
#
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/desktop_jump_route_e2e.txt

click "新建主机"
settle 400
click 500 230
type jump
click 500 300
type 10.0.0.1
click "添加主机"
settle 400

click "新建主机"
settle 300
click "RDP"
settle 300
click 500 170
type desk
click 500 245
type 10.0.0.9
click 500 310
type alice
click "添加主机"
settle 400

# The desktop is the only target a route can reach with one SSH host: the
# route dialog preselects it, and only an SSH host may be its jump.
click "跳板路由"
settle 400
click id:proxies/proxy-add
settle 400
waitfor "desk" 2000
waitfor "jump" 2000
click "添加路由"
settle 400
waitfor "10.0.0.1 → rdp://10.0.0.9" 2000
expect layout clean

# The host form shows the route, and the same-machine switch gives way to it.
click "主机"
settle 400
click 400 60
type desk
settle 300
rightclick 500 160
settle 300
click 560 278
settle 400
waitfor "经由 jump 跳转。" 2000
waitfor "通过 SSH 连接(由该主机的跳板路由接管)" 2000
expect layout clean
shot /tmp/desktop-jump-route-e2e.png
