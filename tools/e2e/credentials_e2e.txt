# End to end: two hosts share one saved password; neither keeps a password of its own,
# and the connection signs in without a password dialog. Against the loopback SSH fixture
# on port 2222 (see tray_e2e.txt for starting it).
#
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/credentials_e2e.txt

click "Settings"
click "Security"
click "Add shared password"
type Lab
click "Password"
type secret
click "Save"
click "Hosts"

click "New Host"
type Lab A
click textfield:Address
type 127.0.0.1
click textfield:User
type tester
click "Password source"
click "Lab"
click textfield:Port
key cmd+a
type 2222
click "Add Host"

dblclick 400 160
waitfor "Trust" 3000
click "Trust and Save"
waitfor "tester@e2e" 8000
expect text "tester@e2e" within 3000
