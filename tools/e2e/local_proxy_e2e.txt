# End to end: open the local proxy panel from the command palette, start a real
# SSH reverse tunnel through e2esshd, run Git HTTPS, pip and npm downloads with
# the copied activation command, restore the shell, and stop the tunnel. The
# fixture must execute the listener inspection with E2E_ALLOW_EXEC=1, and PATH
# must lead with a fake `ss` that verifies and reports the actual loopback
# listener. recording_http_proxy.py on 48997 forwards to the user's HTTP proxy
# on 7897 and logs every CONNECT target.
#
#   (cd tools/e2esshd && go build -o /tmp/e2esshd .)
#   mkdir -p /tmp/motty-proxy-e2e-bin
#   printf '#!/bin/sh\nfor arg do last=$arg; done\nport=${last##*:}\nlsof -nP -a -iTCP:$port -sTCP:LISTEN >/tmp/motty-ss-listeners || exit 1\nif grep -Eq "TCP [^ ]+->" /tmp/motty-ss-listeners || ! grep -Eq "TCP 127[.]0[.]0[.]1:$port " /tmp/motty-ss-listeners; then exit 1; fi\nprintf "LISTEN 0 128 127.0.0.1:%%s 0.0.0.0:*\\n" "$port"\n' > /tmp/motty-proxy-e2e-bin/ss
#   chmod +x /tmp/motty-proxy-e2e-bin/ss
#   E2E_ALLOW_EXEC=1 PATH=/tmp/motty-proxy-e2e-bin:$PATH /tmp/e2esshd 2222 &
#   python3 tools/e2e/recording_http_proxy.py --listen-port 48997 --upstream 127.0.0.1:7897 --log /tmp/motty-proxy.log &
#   make head
#   MOTTY_LANG=en python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/local_proxy_e2e.txt

type download_dir=$(mktemp -d /tmp/motty-local-proxy-downloads.XXXXXX) && route_before=$(route -n get default | awk '/gateway:/{print $2}') && printf "DOWNLOAD_%s=%s ROUTE_BEFORE=%s\n" DIR "$download_dir" "$route_before"
key enter
waitfor "DOWNLOAD_DIR=" 3000
type GIT_CONFIG_GLOBAL=/dev/null git -c http.proxy=http://127.0.0.1:1 ls-remote https://github.com/mattn/go-isatty.git >/dev/null 2>&1 || printf "UNREACHABLE_PROXY_%s\n" GIT_FAILED
key enter
waitfor "UNREACHABLE_PROXY_GIT_FAILED" 5000
type GIT_CONFIG_GLOBAL=/dev/null git -c http.proxy="$https_proxy" clone --depth 1 https://github.com/mattn/go-isatty.git "$download_dir/go-isatty" && printf "GIT_DOWNLOAD_%s\n" OK
key enter
waitfor "GIT_DOWNLOAD_OK" 30000

type pip3 download --disable-pip-version-check --no-cache-dir --no-deps --dest "$download_dir" idna==3.10 && printf "PIP_DOWNLOAD_%s\n" OK

key enter
waitfor "NPM_DOWNLOAD_OK" 60000
type test -d "$download_dir/go-isatty/.git" && test -f "$download_dir/idna-3.10-py3-none-any.whl" && test -f "$download_dir/is-number-7.0.0.tgz" && test "$(shasum -a 256 "$download_dir/idna-3.10-py3-none-any.whl" | awk '{print $1}')" = 946d195a0d259cbba61165e88e65941f16e9b36ea6ddb97f00452bae8b1287d3 && test "$(shasum -a 256 "$download_dir/is-number-7.0.0.tgz" | awk '{print $1}')" = 7b75c1057198cf97696909a9bee176c9c5e9bcb5b03bf3ecef2f484defadd51e && route_now=$(route -n get default | awk '/gateway:/{print $2}') && test "$route_now" = "$route_before" && printf "ARTIFACT_SHA_%s\nROUTE_%s\n" OK UNCHANGED
key enter
